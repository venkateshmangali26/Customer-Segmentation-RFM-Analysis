def generate_customer_profiles(n_customers: int, seed: int = RANDOM_SEED) -> pd.DataFrame:

    assigned_countries = np.random.choice(countries, size=n_customers, p=weights)

def clean_transactions(
    df: pd.DataFrame,
    cap_outliers: bool = True,
    output_path: Optional[Path] = CLEANED_DATA_FILE,
) -> Tuple[pd.DataFrame, Dict]:
    initial_rows = len(df)
    initial_customers = df["CustomerID"].nunique()

    missing_customers = df["CustomerID"].isna().sum()
    pct_missing_customers = (missing_customers / initial_rows) * 100

    df_clean = df.dropna(subset=["CustomerID"]).copy()
    # Safe float-to-int-to-string conversion for customer IDs
    df_clean["CustomerID"] = df_clean["CustomerID"].astype(float).astype(int).astype(str)

    is_cancelled = df_clean["InvoiceNo"].str.startswith("C", na=False) | (df_clean["Quantity"] <= 0)
    cancelled_count = is_cancelled.sum()
    cancellation_rate = (cancelled_count / len(df_clean)) * 100

    valid_mask = (~is_cancelled) & (df_clean["UnitPrice"] > 0)
    df_valid = df_clean[valid_mask].copy()

    df_valid["InvoiceDate"] = pd.to_datetime(df_valid["InvoiceDate"])
    df_valid["TotalPrice"] = df_valid["Quantity"] * df_valid["UnitPrice"]

    outliers_capped = 0
    if cap_outliers:
        q_qty_999 = df_valid["Quantity"].quantile(0.999)
        q_price_999 = df_valid["UnitPrice"].quantile(0.999)
        qty_mask = df_valid["Quantity"] > q_qty_999
        price_mask = df_valid["UnitPrice"] > q_price_999
        outliers_capped = int(qty_mask.sum() + price_mask.sum())

        df_valid.loc[qty_mask, "Quantity"] = q_qty_999
        df_valid.loc[price_mask, "UnitPrice"] = q_price_999
        df_valid["TotalPrice"] = df_valid["Quantity"] * df_valid["UnitPrice"]

    summary = {
        "initial_records": initial_rows,
        "initial_unique_customers": initial_customers,
        "missing_customer_ids": int(missing_customers),
        "pct_missing_customer_ids": round(pct_missing_customers, 2),
        "cancelled_records": int(cancelled_count),
        "cancellation_rate_pct": round(cancellation_rate, 2),
        "retained_clean_records": len(df_valid),
        "retained_unique_customers": df_valid["CustomerID"].nunique(),
        "outliers_capped": outliers_capped,
        "total_revenue": round(float(df_valid["TotalPrice"].sum()), 2),
        "date_range_start": str(df_valid["InvoiceDate"].min()),
        "date_range_end": str(df_valid["InvoiceDate"].max()),
    }

    if output_path:
        output_path = Path(output_path)
        output_path.parent.mkdir(parents=True, exist_ok=True)
        df_valid.to_csv(output_path, index=False)
        print(f"Data cleaned successfully: {len(df_valid):,} valid records saved to {output_path}")

    return df_valid, summary

def calculate_rfm_metrics(
    df: pd.DataFrame,
    snapshot_date: Optional[pd.Timestamp] = None,
) -> pd.DataFrame:
    df = df.copy()
    df["InvoiceDate"] = pd.to_datetime(df["InvoiceDate"])

    if snapshot_date is None:
        snapshot_date = df["InvoiceDate"].max() + pd.Timedelta(days=1)
    else:
        snapshot_date = pd.to_datetime(snapshot_date)

    rfm = df.groupby("CustomerID").agg(
        Recency=("InvoiceDate", lambda d: (snapshot_date - d.max()).days),
        Frequency=("InvoiceNo", "nunique"),
        Monetary=("TotalPrice", "sum"),
        FirstPurchase=("InvoiceDate", "min"),
        LastPurchase=("InvoiceDate", "max"),
        UniqueProducts=("StockCode", "nunique"),
        Country=("Country", lambda c: c.mode()[0] if not c.empty else "Unknown"),
    ).reset_index()

    rfm["AOV"] = (rfm["Monetary"] / rfm["Frequency"]).round(2)
    rfm["Tenure_Days"] = (snapshot_date - rfm["FirstPurchase"]).dt.days
    rfm["Lifespan_Days"] = (rfm["LastPurchase"] - rfm["FirstPurchase"]).dt.days
    rfm["Monetary"] = rfm["Monetary"].round(2)

    return rfm

def score_rfm_quantiles(rfm_df: pd.DataFrame) -> pd.DataFrame:
    df = rfm_df.copy()

    # duplicates='drop' prevents crashes if bin edges are identical due to ties
    df["R_Score"] = pd.qcut(
        df["Recency"].rank(method="first"),
        q=5,
        labels=[5, 4, 3, 2, 1],
        duplicates="drop",
    ).astype(int)

    df["F_Score"] = pd.qcut(
        df["Frequency"].rank(method="first"),
        q=5,
        labels=[1, 2, 3, 4, 5],
        duplicates="drop",
    ).astype(int)

    df["M_Score"] = pd.qcut(
        df["Monetary"].rank(method="first"),
        q=5,
        labels=[1, 2, 3, 4, 5],
        duplicates="drop",
    ).astype(int)

    df["RFM_Score_Code"] = df["R_Score"].astype(str) + df["F_Score"].astype(str) + df["M_Score"].astype(str)
    df["RFM_Score_Total"] = df["R_Score"] + df["F_Score"] + df["M_Score"]

    return df

def assign_rfm_segments(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()

    def get_segment(row) -> str:
        r, f, m = row["R_Score"], row["F_Score"], row["M_Score"]
        if r >= 4 and f >= 4 and m >= 4:
            return "Champions"
        if r <= 2 and f >= 4 and m >= 4:
            return "Can't Lose Them"
        if r >= 3 and f >= 3 and m >= 3:

        if r <= 2 and (f >= 3 or m >= 3):

        if r >= 4 and f == 1:

            return "Promising"
        if r in [2, 3] and f in [2, 3] and m in [2, 3]:
            return "Customers Needing Attention"

def preprocess_rfm_for_clustering(
    rfm_df: pd.DataFrame,
    features: Optional[List[str]] = None,
) -> Tuple[np.ndarray, pd.DataFrame, StandardScaler, Dict]:
    if features is None:
        features = ["Recency", "Frequency", "Monetary"]

    raw_features = rfm_df[features].copy()
    initial_skew = {col: round(float(stats.skew(raw_features[col])), 3) for col in features}

    log_transformed = np.log1p(raw_features)
    post_log_skew = {col: round(float(stats.skew(log_transformed[col])), 3) for col in features}

    scaler = StandardScaler()
    scaled_array = scaler.fit_transform(log_transformed)
    scaled_df = pd.DataFrame(scaled_array, columns=[f"{col}_scaled" for col in features], index=rfm_df.index)

    skew_info = {"features": features, "initial_skewness": initial_skew, "post_log_skewness": post_log_skew}
    return scaled_array, scaled_df, scaler, skew_info

def evaluate_cluster_range(
    scaled_features: np.ndarray,
    k_min: int = K_MIN,
    k_max: int = K_MAX,
    random_state: int = RANDOM_SEED,
) -> pd.DataFrame:
    results = []
    for k in range(k_min, k_max + 1):
        kmeans = KMeans(n_clusters=k, init="k-means++", n_init=10, max_iter=300, random_state=random_state)
        labels = kmeans.fit_predict(scaled_features)

        inertia = kmeans.inertia_
        sample_sz = 1500 if len(scaled_features) > 1500 else None
        sil = silhouette_score(scaled_features, labels, sample_size=sample_sz, random_state=random_state)
        db_index = davies_bouldin_score(scaled_features, labels)
        ch_score = calinski_harabasz_score(scaled_features, labels)

        results.append({
            "k": k,
            "Inertia_WCSS": round(float(inertia), 2),
            "Silhouette_Score": round(float(sil), 4),
            "Davies_Bouldin_Index": round(float(db_index), 4),
            "Calinski_Harabasz_Score": round(float(ch_score), 2),
        })

    eval_df = pd.DataFrame(results)
    eval_file = REPORTS_DIR / "clustering_evaluation_metrics.csv"
    eval_file.parent.mkdir(parents=True, exist_ok=True)
    eval_df.to_csv(eval_file, index=False)
    return eval_df

def train_kmeans_clustering(
    scaled_features: np.ndarray,
    n_clusters: int = DEFAULT_OPTIMAL_K,
    random_state: int = RANDOM_SEED,
) -> Tuple[KMeans, np.ndarray]:
    model = KMeans(n_clusters=n_clusters, init="k-means++", n_init=30, max_iter=500, random_state=random_state)
    cluster_labels = model.fit_predict(scaled_features)
    return model, cluster_labels

def apply_pca_reduction(
    scaled_features: np.ndarray,
    feature_names: List[str],
    n_components: int = 3,
    random_state: int = RANDOM_SEED,
) -> Tuple[np.ndarray, PCA, pd.DataFrame]:
    pca = PCA(n_components=n_components, random_state=random_state)
    pca_coords = pca.fit_transform(scaled_features)
    loadings = pd.DataFrame(pca.components_.T, columns=[f"PC{i+1}" for i in range(n_components)], index=feature_names).round(4)
    return pca_coords, pca, loadings

def simulate_campaign_ab_test(
    rfm_df: pd.DataFrame,
    baseline_response_rate: float = 0.032,
    target_lift_pct: float = 15.0,
    sample_split: float = 0.5,
    random_state: int = 42,
) -> Dict:
    np.random.seed(random_state)
    n_total = len(rfm_df)
    n_control = int(n_total * sample_split)
    n_treatment = n_total - n_control

    control_success = np.random.binomial(n=n_control, p=baseline_response_rate)
    control_rate = control_success / n_control

    targeted_rate_calibrated = baseline_response_rate * (1.0 + (target_lift_pct / 100.0))
    treatment_success = np.random.binomial(n=n_treatment, p=targeted_rate_calibrated)
    treatment_rate = treatment_success / n_treatment

    abs_lift = treatment_rate - control_rate
    rel_lift = (abs_lift / control_rate) * 100.0

    pooled_p = (control_success + treatment_success) / (n_control + n_treatment)
    se = np.sqrt(pooled_p * (1 - pooled_p) * (1 / n_control + 1 / n_treatment))
    z_stat = (treatment_rate - control_rate) / se if se > 0 else 0
    p_val = 2 * (1 - stats.norm.cdf(abs(z_stat)))

    cpc_email = 0.05
    aov_control = 65.00
    aov_targeted = 82.50
    margin = 0.45

    control_cost = n_control * cpc_email
    control_revenue = control_success * aov_control
    control_profit = (control_revenue * margin) - control_cost
    control_roi = ((control_profit) / control_cost) * 100.0 if control_cost > 0 else 0

    treatment_cost = n_treatment * cpc_email
    treatment_revenue = treatment_success * aov_targeted
    treatment_profit = (treatment_revenue * margin) - treatment_cost
    treatment_roi = ((treatment_profit) / treatment_cost) * 100.0 if treatment_cost > 0 else 0

    return {
        "total_audience": n_total,
        "control_sample_size": n_control,
        "treatment_sample_size": n_treatment,
        "control_conversions": int(control_success),
        "treatment_conversions": int(treatment_success),
        "control_response_rate_pct": round(control_rate * 100, 2),
        "treatment_response_rate_pct": round(treatment_rate * 100, 2),
        "absolute_lift_pct_pts": round(abs_lift * 100, 2),
        "relative_lift_pct": round(rel_lift, 2),
        "z_statistic": round(float(z_stat), 3),
        "p_value": round(float(p_val), 4),
        "statistically_significant": bool(p_val < 0.05),
        "control_cost_usd": round(control_cost, 2),
        "control_revenue_usd": round(control_revenue, 2),
        "control_net_profit_usd": round(control_profit, 2),
        "control_roi_pct": round(control_roi, 1),
        "treatment_cost_usd": round(treatment_cost, 2),
        "treatment_revenue_usd": round(treatment_revenue, 2),
        "treatment_net_profit_usd": round(treatment_profit, 2),
        "treatment_roi_pct": round(treatment_roi, 1),
        "profit_gain_usd": round(treatment_profit - control_profit, 2),
    }

from src.marketing_strategy import (
    generate_marketing_playbook_report,
    simulate_campaign_ab_test,
    simulate_full_segmented_rollout,
)
from src.rfm_analysis import run_rfm_pipeline
from src.visualization import generate_all_visualizations

def run_full_pipeline(
    regenerate_data: bool = False,
    n_transactions: int = DEFAULT_N_TRANSACTIONS,
    n_customers: int = DEFAULT_N_CUSTOMERS,
    n_clusters: int = DEFAULT_OPTIMAL_K,
) -> None:
    start_time = time.time()
    print("=" * 80)
    print(" CUSTOMER SEGMENTATION & RFM BEHAVIORAL ANALYSIS PIPELINE")
    print("=" * 80)

    if regenerate_data or not RAW_DATA_FILE.exists():
        print(f"\n[Step 1/6] Generating synthetic transactions ({n_transactions:,}+ records)...")
        generate_synthetic_transactions(n_transactions=n_transactions, n_customers=n_customers, output_path=RAW_DATA_FILE)
    else:
        print(f"\n[Step 1/6] Loading existing raw dataset from {RAW_DATA_FILE}...")

    raw_df = load_raw_data(RAW_DATA_FILE)
    print(f" Loaded {len(raw_df):,} raw transaction rows across {raw_df['CustomerID'].nunique():,} unique customers.")

    print("\n[Step 2/6] Cleaning transactions and removing guest checkouts/anomalies...")
    cleaned_df, cleaning_stats = clean_transactions(raw_df, output_path=CLEANED_DATA_FILE)
    print(f" Cleaned records retained: {cleaning_stats['retained_clean_records']:,}")

    print("\n[Step 3/6] Computing customer-level RFM metrics & quintile scoring...")
    rfm_df, rfm_summary = run_rfm_pipeline(cleaned_df, output_path=RFM_DATA_FILE)
    print(f" Segmented {len(rfm_df):,} customers across RFM segments.")

    print(f"\n[Step 4/6] Training K-Means clustering (k={n_clusters}) & computing PCA projections...")
    rfm_clustered, cluster_profiles, eval_df, pca_model, kmeans_model = run_clustering_pipeline(
        rfm_df, n_clusters=n_clusters, output_path=RFM_DATA_FILE
    )

    print("\n[Step 5/6] Simulating targeted marketing campaigns & A/B test lift (+15% benchmark)...")
    ab_test_res = simulate_campaign_ab_test(rfm_clustered, target_lift_pct=15.0)
    rollout_df = simulate_full_segmented_rollout(rfm_clustered)
    playbook_report = generate_marketing_playbook_report(rfm_clustered)

    report_file = PROJECT_ROOT / "reports" / "marketing_playbook_report.md"
    report_file.parent.mkdir(parents=True, exist_ok=True)
    report_file.write_text(playbook_report, encoding="utf-8")

    print("\n[Step 6/6] Generating publication-grade figures...")
    generate_all_visualizations(rfm_clustered, cluster_profiles, eval_df, rollout_df, output_dir=FIGURES_DIR)

    elapsed = time.time() - start_time
    print("=" * 80)
    print(f" PIPELINE COMPLETED SUCCESSFULLY IN {elapsed:.2f} SECONDS")
    print("=" * 80)

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Run Customer Segmentation & RFM Analysis Pipeline")
    parser.add_argument("--regenerate", action="store_true", help="Force regenerate raw transaction data")
    args = parser.parse_args()
    run_full_pipeline(regenerate_data=args.regenerate)
