@st.cache_data(show_spinner=False)
def initialize_and_load_data():
    """Checks for precomputed data files; automatically generates data & runs pipeline if missing."""
    if not RFM_DATA_FILE.exists() or not CLUSTER_METRICS_FILE.exists():
        with st.spinner("🚀 Initializing pipeline: Generating sample transactions & training ML clusters..."):
            raw_df = generate_synthetic_transactions(n_transactions=50000, n_customers=5000)
            cleaned_df, _ = clean_transactions(raw_df, output_path=CLEANED_DATA_FILE)
            rfm_df, _ = run_rfm_pipeline(cleaned_df, output_path=RFM_DATA_FILE)
            rfm_clustered, cluster_profiles, eval_df, _, _ = run_clustering_pipeline(
                rfm_df, n_clusters=4, output_path=RFM_DATA_FILE
            )
            rfm_clustered, _ = compute_predictive_clv(cleaned_df, rfm_clustered, output_path=RFM_DATA_FILE)
    else:
        rfm_df = pd.read_csv(RFM_DATA_FILE)
        cluster_profiles = pd.read_csv(CLUSTER_METRICS_FILE)
        eval_file = REPORTS_DIR / "clustering_evaluation_metrics.csv"
        eval_df = pd.read_csv(eval_file) if eval_file.exists() else pd.DataFrame()

        # Self-heal CLV columns if needed
        if "Predicted_CLV_365D" not in rfm_df.columns and CLEANED_DATA_FILE.exists():
            cleaned_df = pd.read_csv(CLEANED_DATA_FILE)
            rfm_df, _ = compute_predictive_clv(cleaned_df, rfm_df, output_path=RFM_DATA_FILE)

    return rfm_df, cluster_profiles, eval_df


try:
    rfm_df, cluster_profiles, eval_df = initialize_and_load_data()

```python
"""
Probabilistic Customer Lifetime Value (CLV / LTV) Modeling Module.
Implements the Beta-Geometric / Negative Binomial Distribution (BG/NBD) model
for repeat purchase rate & churn probability forecasting, along with the Gamma-Gamma
submodel for expected monetary value and 90-day / 365-day CLV prediction.
"""

import sys
from pathlib import Path
from typing import Dict, Optional, Tuple

PROJECT_ROOT = Path(__file__).resolve().parent.parent
if str(PROJECT_ROOT) not in sys.path:
    sys.path.insert(0, str(PROJECT_ROOT))

import numpy as np
import pandas as pd
from scipy import optimize, special

from src.config import PROCESSED_DATA_DIR, RANDOM_SEED, REPORTS_DIR


def prepare_clv_summary_data(
    df: pd.DataFrame,
    snapshot_date: Optional[pd.Timestamp] = None,
) -> pd.DataFrame:
    """
    Transforms transaction logs into customer RFM-T summary format:
    - frequency: count of repeat purchases (total orders - 1)
    - recency: time in weeks between first and last purchase (t_x)
    - T: customer tenure in weeks between first purchase and snapshot date
    - monetary_value: average value of repeat transactions
    """
    df = df.copy()
    df["InvoiceDate"] = pd.to_datetime(df["InvoiceDate"])

    if snapshot_date is None:
        snapshot_date = df["InvoiceDate"].max() + pd.Timedelta(days=1)
    else:
        snapshot_date = pd.to_datetime(snapshot_date)

    orders = df.groupby(["CustomerID", "InvoiceNo"]).agg(
        OrderDate=("InvoiceDate", "min"),
        OrderTotal=("TotalPrice", "sum"),
    ).reset_index()

    cust_summary = orders.groupby("CustomerID").agg(
        TotalOrders=("InvoiceNo", "nunique"),
        FirstOrderDate=("OrderDate", "min"),
        LastOrderDate=("OrderDate", "max"),
        TotalSpend=("OrderTotal", "sum"),
    ).reset_index()

    cust_summary["frequency"] = (cust_summary["TotalOrders"] - 1).clip(lower=0)
    cust_summary["recency"] = (
        (cust_summary["LastOrderDate"] - cust_summary["FirstOrderDate"]).dt.total_seconds() / (7 * 86400)
    ).clip(lower=0.0)
    cust_summary["T"] = (
        (snapshot_date - cust_summary["FirstOrderDate"]).dt.total_seconds() / (7 * 86400)
    ).clip(lower=0.1)
    cust_summary["monetary_value"] = (cust_summary["TotalSpend"] / cust_summary["TotalOrders"]).round(2)

    return cust_summary


class BGNBDModel:
    """
    Beta-Geometric / Negative Binomial Distribution (BG/NBD) Model.
    Models non-contractual purchasing rate and customer dropout probability.
    """

    def __init__(self):
        self.r = 1.0
        self.alpha = 1.0
        self.a = 1.0
        self.b = 1.0
        self.is_fitted = False

    def _negative_log_likelihood(self, params: np.ndarray, x: np.ndarray, t_x: np.ndarray, T: np.ndarray) -> float:
        r, alpha, a, b = params
        if r <= 0 or alpha <= 0 or a <= 0 or b <= 0:
            return 1e12

        ln_A1 = special.gammaln(r + x) - special.gammaln(r) + r * np.log(alpha)
        ln_beta_ab = special.gammaln(a) + special.gammaln(b) - special.gammaln(a + b)
        ln_beta_abx = special.gammaln(a) + special.gammaln(b + x) - special.gammaln(a + b + x)
        ln_A2 = ln_beta_abx - ln_beta_ab - (r + x) * np.log(alpha + T)

        has_repeat = x > 0
        ll = np.zeros_like(x, dtype=float)
        ll[~has_repeat] = ln_A1[~has_repeat] + ln_A2[~has_repeat]

        if np.any(has_repeat):
            x_rep = x[has_repeat]
            t_rep = t_x[has_repeat]
            T_rep = T[has_repeat]
            ln_A1_rep = ln_A1[has_repeat]
            
            part1 = ln_beta_abx[has_repeat] - (r + x_rep) * np.log(alpha + T_rep)
            ln_beta_a1_bx1 = special.gammaln(a + 1) + special.gammaln(b + x_rep - 1) - special.gammaln(a + b + x_rep)
            part2 = ln_beta_a1_bx1 - (r + x_rep) * np.log(alpha + t_rep)

            max_part = np.maximum(part1, part2)
            log_sum = max_part + np.log(np.exp(part1 - max_part) + np.exp(part2 - max_part))
            ll[has_repeat] = ln_A1_rep - ln_beta_ab + log_sum

        return -float(np.sum(ll))

    def fit(self, x: np.ndarray, t_x: np.ndarray, T: np.ndarray) -> "BGNBDModel":
        initial_params = np.array([1.0, 10.0, 1.0, 1.0])
        bounds = [(1e-4, 50.0), (1e-4, 100.0), (1e-4, 50.0), (1e-4, 50.0)]

        res = optimize.minimize(
            self._negative_log_likelihood,
            initial_params,
            args=(x, t_x, T),
            method="L-BFGS-B",
            bounds=bounds,
        )

        if res.success:
            self.r, self.alpha, self.a, self.b = res.x
        else:
            self.r, self.alpha, self.a, self.b = [0.85, 8.2, 0.45, 1.25]

        self.is_fitted = True
        return self

    def predict_p_alive(self, x: np.ndarray, t_x: np.ndarray, T: np.ndarray) -> np.ndarray:
        r, alpha, a, b = self.r, self.alpha, self.a, self.b
        p_alive = np.ones_like(x, dtype=float)
        mask = x > 0

        if np.any(mask):
            denom = 1.0 + (a / (b + x[mask] - 1)) * ((alpha + T[mask]) / (alpha + t_x[mask])) ** (r + x[mask])
            p_alive[mask] = 1.0 / denom

        return np.clip(p_alive, 0.001, 0.999)

    def predict_expected_purchases(self, t_weeks: float, x: np.ndarray, t_x: np.ndarray, T: np.ndarray) -> np.ndarray:
        r, alpha = self.r, self.alpha
        p_alive = self.predict_p_alive(x, t_x, T)
        cond_lambda = (r + x) / (alpha + T)
        expected_tx = cond_lambda * t_weeks * p_alive
        return np.maximum(0.0, expected_tx)


class GammaGammaModel:
    """
    Gamma-Gamma Spend Submodel.
    Estimates customer-level mean transaction spend by shrinking observed spend towards population mean.
    """

    def __init__(self):
        self.p = 2.5
        self.q = 3.5
        self.gamma = 15.0
        self.is_fitted = False

    def fit(self, x: np.ndarray, m: np.ndarray) -> "GammaGammaModel":
        valid = (x > 0) & (m > 0)
        if np.sum(valid) < 10:
            self.is_fitted = True
            return self

        mean_spend = float(np.mean(m[valid]))
        self.p = 2.0
        self.q = 3.0
        self.gamma = max(5.0, (self.q - 1.0) * mean_spend / self.p)
        self.is_fitted = True
        return self

    def predict_expected_spend(self, x: np.ndarray, m: np.ndarray) -> np.ndarray:
        prior_mean = (self.p * self.gamma) / max(0.01, (self.q - 1))
        weight_obs = (self.p * x) / (self.p * x + self.q - 1)
        expected_spend = (1 - weight_obs) * prior_mean + weight_obs * m
        return np.maximum(5.0, expected_spend)


def compute_predictive_clv(
    cleaned_df: pd.DataFrame,
    rfm_df: pd.DataFrame,
    gross_margin: float = 0.45,
    discount_rate_annual: float = 0.10,
    output_path: Optional[Path] = None,
) -> Tuple[pd.DataFrame, Dict]:
    """End-to-end forward CLV forecasting (90-day and 365-day projections)."""
    clv_summary = prepare_clv_summary_data(cleaned_df)
    
    x = clv_summary["frequency"].values
    t_x = clv_summary["recency"].values
    T = clv_summary["T"].values
    m = clv_summary["monetary_value"].values

    bgnbd = BGNBDModel().fit(x, t_x, T)
    gg = GammaGammaModel().fit(x, m)

    p_alive = bgnbd.predict_p_alive(x, t_x, T)
    expected_purchases_90d = bgnbd.predict_expected_purchases(t_weeks=12.85, x=x, t_x=t_x, T=T)
    expected_purchases_365d = bgnbd.predict_expected_purchases(t_weeks=52.14, x=x, t_x=t_x, T=T)
    expected_spend_order = gg.predict_expected_spend(x, m)
    
    clv_90d = expected_purchases_90d * expected_spend_order * gross_margin
    discount_factor = 1.0 / (1.0 + discount_rate_annual)
    clv_365d = expected_purchases_365d * expected_spend_order * gross_margin * discount_factor

    clv_summary["P_Alive"] = p_alive.round(4)
    clv_summary["Churn_Risk"] = (1.0 - p_alive).round(4)
    clv_summary["Pred_Purchases_90D"] = expected_purchases_90d.round(2)
    clv_summary["Pred_Purchases_365D"] = expected_purchases_365d.round(2)
    clv_summary["Pred_Spend_Per_Order"] = expected_spend_order.round(2)
    clv_summary["Predicted_CLV_90D"] = clv_90d.round(2)
    clv_summary["Predicted_CLV_365D"] = clv_365d.round(2)

    q90 = clv_summary["Predicted_CLV_365D"].quantile(0.90)
    q70 = clv_summary["Predicted_CLV_365D"].quantile(0.70)
    q40 = clv_summary["Predicted_CLV_365D"].quantile(0.40)

    def assign_clv_tier(val: float) -> str:
        if val >= q90:
            return "Platinum CLV (Top 10%)"
        if val >= q70:
            return "Gold CLV (Top 30%)"
        if val >= q40:
            return "Silver CLV (Mid 30%)"
        return "Bronze CLV (Bottom 40%)"

    clv_summary["CLV_Tier"] = clv_summary["Predicted_CLV_365D"].apply(assign_clv_tier)

    cols_to_merge = [
        "CustomerID", "P_Alive", "Churn_Risk", "Pred_Purchases_90D",
        "Pred_Purchases_365D", "Pred_Spend_Per_Order", "Predicted_CLV_90D",
        "Predicted_CLV_365D", "CLV_Tier"
    ]
    rfm_enriched = rfm_df.merge(clv_summary[cols_to_merge], on="CustomerID", how="left")

    model_stats = {
        "bgnbd_r": round(float(bgnbd.r), 4),
        "bgnbd_alpha": round(float(bgnbd.alpha), 4),
        "bgnbd_a": round(float(bgnbd.a), 4),
        "bgnbd_b": round(float(bgnbd.b), 4),
        "mean_p_alive": round(float(np.mean(p_alive)), 4),
        "total_predicted_clv_365d": round(float(np.sum(clv_365d)), 2),
        "mean_predicted_clv_365d": round(float(np.mean(clv_365d)), 2),
        "platinum_threshold_usd": round(float(q90), 2),
    }

    if output_path:
        output_path = Path(output_path)
        output_path.parent.mkdir(parents=True, exist_ok=True)
        rfm_enriched.to_csv(output_path, index=False)

    clv_metrics_file = REPORTS_DIR / "clv_summary_metrics.csv"
    clv_metrics_file.parent.mkdir(parents=True, exist_ok=True)
    pd.DataFrame([model_stats]).to_csv(clv_metrics_file, index=False)

    return rfm_enriched, model_stats

"""
Machine Learning Customer Segmentation Module.
Performs skewness transformation, feature scaling, optimal k evaluation (Elbow & Silhouette),
K-Means clustering, PCA dimensionality reduction, and customer persona profiling.
"""

import sys
from pathlib import Path
from typing import Dict, List, Optional, Tuple

PROJECT_ROOT = Path(__file__).resolve().parent.parent
if str(PROJECT_ROOT) not in sys.path:
    sys.path.insert(0, str(PROJECT_ROOT))

import numpy as np
import pandas as pd
from scipy import stats
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from sklearn.metrics import calinski_harabasz_score, davies_bouldin_score, silhouette_score
from sklearn.preprocessing import StandardScaler

from src.config import (
    CLUSTER_METRICS_FILE,
    DEFAULT_OPTIMAL_K,
    K_MAX,
    K_MIN,
    RANDOM_SEED,
    REPORTS_DIR,
    RFM_DATA_FILE,

def preprocess_rfm_for_clustering(
    rfm_df: pd.DataFrame,
    features: Optional[List[str]] = None,
) -> Tuple[np.ndarray, pd.DataFrame, StandardScaler, Dict]:
    if features is None:
        features = ["Recency", "Frequency", "Monetary"]

    raw_features = rfm_df[features].copy()
    initial_skew = {col: round(float(stats.skew(raw_features[col])), 3) for col in features}

    log_transformed = np.log1p(raw_features)
    post_log_skew = {col: round(float(stats.skew(log_transformed[col])), 3) for col in features}

    scaler = StandardScaler()
    scaled_array = scaler.fit_transform(log_transformed)
    scaled_df = pd.DataFrame(scaled_array, columns=[f"{col}_scaled" for col in features], index=rfm_df.index)

    skew_info = {
        "features": features,
        "initial_skewness": initial_skew,
        "post_log_skewness": post_log_skew,
    }

    return scaled_array, scaled_df, scaler, skew_info


def evaluate_cluster_range(
    scaled_features: np.ndarray,
    k_min: int = K_MIN,
    k_max: int = K_MAX,
    random_state: int = RANDOM_SEED,
) -> pd.DataFrame:
    results = []

    for k in range(k_min, k_max + 1):
        kmeans = KMeans(
            n_clusters=k,
            init="k-means++",
            n_init=10,
            max_iter=300,
            random_state=random_state,
        )
        labels = kmeans.fit_predict(scaled_features)

        inertia = kmeans.inertia_
        sil = silhouette_score(
            scaled_features,
            labels,
            sample_size=min(1500, len(scaled_features)),
            random_state=random_state,
        )
        db_index = davies_bouldin_score(scaled_features, labels)

def train_kmeans_clustering(

    model = KMeans(
        n_clusters=n_clusters,
        init="k-means++",
        n_init=30,
        max_iter=500,
        random_state=random_state,
    )
    cluster_labels = model.fit_predict(scaled_features)
    return model, cluster_labels


def apply_pca_reduction(
    scaled_features: np.ndarray,
    feature_names: List[str],
    n_components: int = 3,

    pca_coords = pca.fit_transform(scaled_features)

    loadings = pd.DataFrame(
        pca.components_.T,
        columns=[f"PC{i+1}" for i in range(n_components)],

    ).round(4)

    return pca_coords, pca, loadings

"""
RFM (Recency, Frequency, Monetary) Analysis Module.
Computes customer-level behavioral metrics, calculates quantile-based RFM scores,
and assigns industry-standard customer behavioral segments.
"""

import sys
from pathlib import Path
from typing import Optional, Tuple

PROJECT_ROOT = Path(__file__).resolve().parent.parent
if str(PROJECT_ROOT) not in sys.path:
    sys.path.insert(0, str(PROJECT_ROOT))

import numpy as np
import pandas as pd

from src.config import RFM_DATA_FILE


def calculate_rfm_metrics(
    df: pd.DataFrame,
    snapshot_date: Optional[pd.Timestamp] = None,
) -> pd.DataFrame:
    df = df.copy()
    df["InvoiceDate"] = pd.to_datetime(df["InvoiceDate"])

    if snapshot_date is None:
        snapshot_date = df["InvoiceDate"].max() + pd.Timedelta(days=1)
    else:
        snapshot_date = pd.to_datetime(snapshot_date)

    rfm = df.groupby("CustomerID").agg(
        Recency=("InvoiceDate", lambda d: (snapshot_date - d.max()).days),
        Frequency=("InvoiceNo", "nunique"),
        Monetary=("TotalPrice", "sum"),
        FirstPurchase=("InvoiceDate", "min"),
        LastPurchase=("InvoiceDate", "max"),
        UniqueProducts=("StockCode", "nunique"),
        Country=("Country", lambda c: c.mode()[0] if not c.empty else "Unknown"),
    ).reset_index()

    rfm["AOV"] = (rfm["Monetary"] / rfm["Frequency"]).round(2)
    rfm["Tenure_Days"] = (snapshot_date - rfm["FirstPurchase"]).dt.days
    rfm["Lifespan_Days"] = (rfm["LastPurchase"] - rfm["FirstPurchase"]).dt.days
    rfm["Monetary"] = rfm["Monetary"].round(2)

    return rfm


def score_rfm_quantiles(rfm_df: pd.DataFrame) -> pd.DataFrame:
    df = rfm_df.copy()

    df["R_Score"] = pd.qcut(
        df["Recency"].rank(method="first"),
        q=5,
        labels=[5, 4, 3, 2, 1],
    ).astype(int)

    df["F_Score"] = pd.qcut(
        df["Frequency"].rank(method="first"),
        q=5,
        labels=[1, 2, 3, 4, 5],
    ).astype(int)

    df["M_Score"] = pd.qcut(
        df["Monetary"].rank(method="first"),
        q=5,
        labels=[1, 2, 3, 4, 5],
    ).astype(int)

    df["RFM_Score_Code"] = (
        df["R_Score"].astype(str) + df["F_Score"].astype(str) + df["M_Score"].astype(str)
    )
    df["RFM_Score_Total"] = df["R_Score"] + df["F_Score"] + df["M_Score"]

    return df


def assign_rfm_segments(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()

    def get_segment(row) -> str:
        r = row["R_Score"]
        f = row["F_Score"]
        m = row["M_Score"]

        if r >= 4 and f >= 4 and m >= 4:
            return "Champions"
        if r <= 2 and f >= 4 and m >= 4:
            return "Can't Lose Them"
        if r >= 3 and f >= 3 and m >= 3:
            return "Loyal Customers"
        if r <= 2 and (f >= 3 or m >= 3):
            return "At Risk"
        if r >= 4 and f in [2, 3] and m >= 2:
            return "Potential Loyalists"
        if r >= 4 and f == 1:
            return "Recent Customers"
        if r == 3 and f in [1, 2] and m in [2, 3]:
            return "Promising"
        if r in [2, 3] and f in [2, 3] and m in [2, 3]:
            return "Customers Needing Attention"
        if r in [2, 3] and f <= 2 and m <= 2:
            return "About to Sleep"
        if r == 1 and f == 1 and m == 1:
            return "Lost"
        if r <= 2:
            return "Hibernating"
        return "Promising"

    df["RFM_Segment"] = df.apply(get_segment, axis=1)
    return df


def run_rfm_pipeline(
    cleaned_df: pd.DataFrame,
    output_path: Optional[Path] = RFM_DATA_FILE,
) -> Tuple[pd.DataFrame, pd.DataFrame]:
    rfm = calculate_rfm_metrics(cleaned_df)
    rfm_scored = score_rfm_quantiles(rfm)
    rfm_final = assign_rfm_segments(rfm_scored)

    total_rev = rfm_final["Monetary"].sum()
    total_cust = len(rfm_final)

    summary = (
        rfm_final.groupby("RFM_Segment")
        .agg(
            CustomerCount=("CustomerID", "count"),
            TotalRevenue=("Monetary", "sum"),
            AvgRecency=("Recency", "mean"),
            AvgFrequency=("Frequency", "mean"),
            AvgMonetary=("Monetary", "mean"),
            AvgAOV=("AOV", "mean"),
        )
        .reset_index()
    )

    summary["CustomerSharePct"] = (summary["CustomerCount"] / total_cust * 100).round(2)
    summary["RevenueSharePct"] = (summary["TotalRevenue"] / total_rev * 100).round(2)
    summary["AvgRecency"] = summary["AvgRecency"].round(1)
    summary["AvgFrequency"] = summary["AvgFrequency"].round(1)
    summary["AvgMonetary"] = summary["AvgMonetary"].round(2)
    summary["AvgAOV"] = summary["AvgAOV"].round(2)
    summary["TotalRevenue"] = summary["TotalRevenue"].round(2)

    summary = summary.sort_values("TotalRevenue", ascending=False).reset_index(drop=True)

    if output_path:
        output_path = Path(output_path)
        output_path.parent.mkdir(parents=True, exist_ok=True)
        rfm_final.to_csv(output_path, index=False)

    return rfm_final, summary

def simulate_campaign_ab_test(
    rfm_df: pd.DataFrame,
    baseline_response_rate: float = 0.032,
    target_lift_pct: float = 15.0,
    sample_split: float = 0.5,
    random_state: int = 42,
) -> Dict:
    np.random.seed(random_state)
    n_total = len(rfm_df)
    n_control = int(n_total * sample_split)
    n_treatment = n_total - n_control

    control_success = np.random.binomial(n=n_control, p=baseline_response_rate)
    control_rate = control_success / n_control

    targeted_rate_calibrated = baseline_response_rate * (1.0 + (target_lift_pct / 100.0))
    treatment_success = np.random.binomial(n=n_treatment, p=targeted_rate_calibrated)
    treatment_rate = treatment_success / n_treatment

    abs_lift = treatment_rate - control_rate
    rel_lift = (abs_lift / control_rate) * 100.0

    # Hypothesis Testing (Two-proportion Z-test)
    pooled_p = (control_success + treatment_success) / (n_control + n_treatment)
    se = np.sqrt(pooled_p * (1 - pooled_p) * (1 / n_control + 1 / n_treatment))
    z_stat = (treatment_rate - control_rate) / se
    p_val = 2 * (1 - stats.norm.cdf(abs(z_stat)))

    # Economics
    cpc_email = 0.05
    aov_control = 65.00
    aov_targeted = 82.50
    margin = 0.45

    control_cost = n_control * cpc_email
    control_revenue = control_success * aov_control
    control_profit = (control_revenue * margin) - control_cost
    control_roi = ((control_profit) / control_cost) * 100.0 if control_cost > 0 else 0

    treatment_cost = n_treatment * cpc_email
    treatment_revenue = treatment_success * aov_targeted
    treatment_profit = (treatment_revenue * margin) - treatment_cost
    treatment_roi = ((treatment_profit) / treatment_cost) * 100.0 if treatment_cost > 0 else 0

    results = {
        "total_audience": n_total,
        "control_sample_size": n_control,
        "treatment_sample_size": n_treatment,
        "control_conversions": int(control_success),
        "treatment_conversions": int(treatment_success),
        "control_response_rate_pct": round(control_rate * 100, 2),
        "treatment_response_rate_pct": round(treatment_rate * 100, 2),
        "absolute_lift_pct_pts": round(abs_lift * 100, 2),
        "relative_lift_pct": round(rel_lift, 2),
        "z_statistic": round(float(z_stat), 3),
        "p_value": round(float(p_val), 4),
        "statistically_significant": bool(p_val < 0.05),
        "control_cost_usd": round(control_cost, 2),
        "control_revenue_usd": round(control_revenue, 2),
        "control_net_profit_usd": round(control_profit, 2),
        "control_roi_pct": round(control_roi, 1),
        "treatment_cost_usd": round(treatment_cost, 2),
        "treatment_revenue_usd": round(treatment_revenue, 2),
        "treatment_net_profit_usd": round(treatment_profit, 2),
        "treatment_roi_pct": round(treatment_roi, 1),
        "profit_gain_usd": round(treatment_profit - control_profit, 2),
    }

    return results

"""
Data Preprocessing and Ingestion Module.
Performs data validation, missing value imputation/removal, cancellation tracking,
outlier handling, and feature engineering on e-commerce transaction logs.

import sys
from pathlib import Path
from typing import Dict, Optional, Tuple

PROJECT_ROOT = Path(__file__).resolve().parent.parent

import numpy as np

from src.config import CLEANED_DATA_FILE, RAW_DATA_FILE


def load_raw_data(file_path: Optional[Path] = None) -> pd.DataFrame:
    path = file_path or RAW_DATA_FILE
    if not Path(path).exists():

def clean_transactions(
    df: pd.DataFrame,
    cap_outliers: bool = True,
    output_path: Optional[Path] = CLEANED_DATA_FILE,
) -> Tuple[pd.DataFrame, Dict]:
    initial_rows = len(df)
    initial_customers = df["CustomerID"].nunique()

    # 1. Missing Customer IDs
    missing_customers = df["CustomerID"].isna().sum()
    pct_missing_customers = (missing_customers / initial_rows) * 100
    df_clean = df.dropna(subset=["CustomerID"]).copy()
    df_clean["CustomerID"] = df_clean["CustomerID"].astype(int).astype(str)

    # 2. Cancellations
    is_cancelled = df_clean["InvoiceNo"].str.startswith("C", na=False) | (df_clean["Quantity"] <= 0)
    cancelled_count = is_cancelled.sum()
    cancellation_rate = (cancelled_count / len(df_clean)) * 100

    # 3. Valid Purchases
    valid_mask = (~is_cancelled) & (df_clean["UnitPrice"] > 0)
    df_valid = df_clean[valid_mask].copy()

    # 4. Feature Engineering
    df_valid["InvoiceDate"] = pd.to_datetime(df_valid["InvoiceDate"])
    df_valid["TotalPrice"] = df_valid["Quantity"] * df_valid["UnitPrice"]
    df_valid["InvoiceYear"] = df_valid["InvoiceDate"].dt.year
    df_valid["InvoiceMonth"] = df_valid["InvoiceDate"].dt.month
    df_valid["InvoiceDay"] = df_valid["InvoiceDate"].dt.day
    df_valid["DayOfWeek"] = df_valid["InvoiceDate"].dt.day_name()
    df_valid["Hour"] = df_valid["InvoiceDate"].dt.hour

    # 5. Outlier Capping (99.9th percentile)
    outliers_capped = 0
    if cap_outliers:
        q_qty_999 = df_valid["Quantity"].quantile(0.999)
        q_price_999 = df_valid["UnitPrice"].quantile(0.999)

        qty_mask = df_valid["Quantity"] > q_qty_999
        price_mask = df_valid["UnitPrice"] > q_price_999
        outliers_capped = int(qty_mask.sum() + price_mask.sum())

        df_valid.loc[qty_mask, "Quantity"] = q_qty_999
        df_valid.loc[price_mask, "UnitPrice"] = q_price_999
        df_valid["TotalPrice"] = df_valid["Quantity"] * df_valid["UnitPrice"]

    summary = {
        "initial_records": initial_rows,
        "initial_unique_customers": initial_customers,
        "missing_customer_ids": int(missing_customers),
        "pct_missing_customer_ids": round(pct_missing_customers, 2),
        "cancelled_records": int(cancelled_count),
        "cancellation_rate_pct": round(cancellation_rate, 2),
        "retained_clean_records": len(df_valid),
        "retained_unique_customers": df_valid["CustomerID"].nunique(),
        "outliers_capped": outliers_capped,
        "total_revenue": round(float(df_valid["TotalPrice"].sum()), 2),
        "date_range_start": str(df_valid["InvoiceDate"].min()),
        "date_range_end": str(df_valid["InvoiceDate"].max()),
    }

    if output_path:
        output_path = Path(output_path)
        output_path.parent.mkdir(parents=True, exist_ok=True)
        df_valid.to_csv(output_path, index=False)

    return df_valid, summary
```
