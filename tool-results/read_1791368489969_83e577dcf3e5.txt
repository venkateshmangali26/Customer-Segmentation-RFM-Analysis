   597→### 2. [`src/data_processing.py`](file:///c:/Customer%20Segmentation%20&%20RFM%20Analysis/src/data_processing.py)
   598→*Data validation, removal of missing customer IDs, cancellation filtering, total price engineering, and outlier capping.*
   599→
   600→```python
   601→"""
   602→Data Preprocessing and Ingestion Module.
   603→Performs data validation, missing value imputation/removal, cancellation tracking,
   604→outlier handling, and feature engineering on e-commerce transaction logs.
   605→"""
   606→
   607→import sys
   608→from pathlib import Path
   609→from typing import Dict, Optional, Tuple
   610→
   611→PROJECT_ROOT = Path(__file__).resolve().parent.parent
   612→if str(PROJECT_ROOT) not in sys.path:
   613→    sys.path.insert(0, str(PROJECT_ROOT))
   614→
   615→import numpy as np
   616→import pandas as pd
   617→
   618→from src.config import CLEANED_DATA_FILE, RAW_DATA_FILE
   619→
   620→
   621→def load_raw_data(file_path: Optional[Path] = None) -> pd.DataFrame:
   622→    """Loads raw transactions CSV."""
   623→    path = file_path or RAW_DATA_FILE
   624→    if not Path(path).exists():
   625→        raise FileNotFoundError(f"Raw data file not found at: {path}")
   626→    df = pd.read_csv(path, dtype={"InvoiceNo": str, "StockCode": str})
   627→    return df
   628→
   629→
   630→def clean_transactions(
   631→    df: pd.DataFrame,
   632→    cap_outliers: bool = True,
   633→    output_path: Optional[Path] = CLEANED_DATA_FILE,
   634→) -> Tuple[pd.DataFrame, Dict]:
   635→    """
   636→    Cleans raw transactions:
   637→    1. Removes missing CustomerIDs.
   638→    2. Isolates and logs cancellations.
   639→    3. Filters valid transactions (Quantity > 0, UnitPrice > 0).
   640→    4. Computes TotalPrice = Quantity * UnitPrice.
   641→    5. Optionally caps extreme outliers at the 99.9th percentile.
   642→    """
   643→    initial_rows = len(df)
   644→    initial_customers = df["CustomerID"].nunique()
   645→
   646→    # 1. Check Missing Values
   647→    missing_customers = df["CustomerID"].isna().sum()
   648→    pct_missing_customers = (missing_customers / initial_rows) * 100
   649→
   650→    # Retain only records with known CustomerID
   651→    df_clean = df.dropna(subset=["CustomerID"]).copy()
   652→    df_clean["CustomerID"] = df_clean["CustomerID"].astype(int).astype(str)
   653→
   654→    # 2. Cancellations & Negative Quantities
   655→    is_cancelled = df_clean["InvoiceNo"].str.startswith("C", na=False) | (df_clean["Quantity"] <= 0)
   656→    cancelled_count = is_cancelled.sum()
   657→    cancellation_rate = (cancelled_count / len(df_clean)) * 100
   658→
   659→    # 3. Filter valid purchases
   660→    valid_mask = (~is_cancelled) & (df_clean["UnitPrice"] > 0)
   661→    df_valid = df_clean[valid_mask].copy()
   662→
   663→    # 4. Date Parsing and Feature Engineering
   664→    df_valid["InvoiceDate"] = pd.to_datetime(df_valid["InvoiceDate"])
   665→    df_valid["TotalPrice"] = df_valid["Quantity"] * df_valid["UnitPrice"]
   666→
   667→    df_valid["InvoiceYear"] = df_valid["InvoiceDate"].dt.year
   668→    df_valid["InvoiceMonth"] = df_valid["InvoiceDate"].dt.month
   669→    df_valid["InvoiceDay"] = df_valid["InvoiceDate"].dt.day
   670→    df_valid["DayOfWeek"] = df_valid["InvoiceDate"].dt.day_name()
   671→    df_valid["Hour"] = df_valid["InvoiceDate"].dt.hour
   672→
   673→    # 5. Outlier Handling (optional capping)
   674→    outliers_capped = 0
   675→    if cap_outliers:
   676→        q_qty_999 = df_valid["Quantity"].quantile(0.999)
   677→        q_price_999 = df_valid["UnitPrice"].quantile(0.999)
   678→
   679→        qty_mask = df_valid["Quantity"] > q_qty_999
   680→        price_mask = df_valid["UnitPrice"] > q_price_999
   681→        outliers_capped = int(qty_mask.sum() + price_mask.sum())
   682→
   683→        df_valid.loc[qty_mask, "Quantity"] = q_qty_999
   684→        df_valid.loc[price_mask, "UnitPrice"] = q_price_999
   685→        df_valid["TotalPrice"] = df_valid["Quantity"] * df_valid["UnitPrice"]
   686→
   687→    # Summary Statistics
   688→    summary = {
   689→        "initial_records": initial_rows,
   690→        "initial_unique_customers": initial_customers,
   691→        "missing_customer_ids": int(missing_customers),
   692→        "pct_missing_customer_ids": round(pct_missing_customers, 2),
   693→        "cancelled_records": int(cancelled_count),
   694→        "cancellation_rate_pct": round(cancellation_rate, 2),
   695→        "retained_clean_records": len(df_valid),
   696→        "retained_unique_customers": df_valid["CustomerID"].nunique(),
   697→        "outliers_capped": outliers_capped,
   698→        "total_revenue": round(float(df_valid["TotalPrice"].sum()), 2),
   699→        "date_range_start": str(df_valid["InvoiceDate"].min()),
   700→        "date_range_end": str(df_valid["InvoiceDate"].max()),
   701→    }
   702→
   703→    if output_path:
   704→        output_path = Path(output_path)
   705→        output_path.parent.mkdir(parents=True, exist_ok=True)
   706→        df_valid.to_csv(output_path, index=False)
   707→        print(f"Data cleaned successfully: {len(df_valid):,} valid records saved to {output_path}")
   708→
   709→    return df_valid, summary
   710→
   711→
   712→if __name__ == "__main__":
   713→    raw_df = load_raw_data()
   714→    cleaned_df, stats = clean_transactions(raw_df)
   715→    for k, v in stats.items():
   716→        print(f"  - {k}: {v}")
   717→```

   721→### 3. [`src/rfm_analysis.py`](file:///c:/Customer%20Segmentation%20&%20RFM%20Analysis/src/rfm_analysis.py)
   722→*Customer-level aggregation of Recency, Frequency, Monetary, Average Order Value (AOV), robust quintile binning (`rank(method='first')`), and 11 business segments.*
   723→
   724→```python
   725→"""
   726→RFM (Recency, Frequency, Monetary) Analysis Module.
   727→Computes customer-level behavioral metrics, calculates quantile-based RFM scores,
   728→and assigns industry-standard customer behavioral segments.
   729→"""
   730→
   731→import sys
   732→from pathlib import Path
   733→from typing import Optional, Tuple
   734→
   735→PROJECT_ROOT = Path(__file__).resolve().parent.parent
   736→if str(PROJECT_ROOT) not in sys.path:
   737→    sys.path.insert(0, str(PROJECT_ROOT))
   738→
   739→import numpy as np
   740→import pandas as pd
   741→
   742→from src.config import RFM_DATA_FILE
   743→
   744→
   745→def calculate_rfm_metrics(
   746→    df: pd.DataFrame,
   747→    snapshot_date: Optional[pd.Timestamp] = None,
   748→) -> pd.DataFrame:
   749→    """Aggregates cleaned transactions into customer-level RFM metrics."""
   750→    df = df.copy()
   751→    df["InvoiceDate"] = pd.to_datetime(df["InvoiceDate"])
   752→
   753→    if snapshot_date is None:
   754→        snapshot_date = df["InvoiceDate"].max() + pd.Timedelta(days=1)
   755→    else:
   756→        snapshot_date = pd.to_datetime(snapshot_date)
   757→
   758→    rfm = df.groupby("CustomerID").agg(
   759→        Recency=("InvoiceDate", lambda d: (snapshot_date - d.max()).days),
   760→        Frequency=("InvoiceNo", "nunique"),
   761→        Monetary=("TotalPrice", "sum"),
   762→        FirstPurchase=("InvoiceDate", "min"),
   763→        LastPurchase=("InvoiceDate", "max"),
   764→        UniqueProducts=("StockCode", "nunique"),
   765→        Country=("Country", lambda c: c.mode()[0] if not c.empty else "Unknown"),
   766→    ).reset_index()
   767→
   768→    rfm["AOV"] = (rfm["Monetary"] / rfm["Frequency"]).round(2)
   769→    rfm["Tenure_Days"] = (snapshot_date - rfm["FirstPurchase"]).dt.days
   770→    rfm["Lifespan_Days"] = (rfm["LastPurchase"] - rfm["FirstPurchase"]).dt.days
   771→    rfm["Monetary"] = rfm["Monetary"].round(2)
   772→
   773→    return rfm
   774→
   775→
   776→def score_rfm_quantiles(rfm_df: pd.DataFrame) -> pd.DataFrame:
   777→    """
   778→    Assigns quintile scores (1 to 5) for Recency, Frequency, and Monetary.
   779→    - Recency: 5 is best (most recent / lowest days).
   780→    - Frequency: 5 is best (highest order count).
   781→    - Monetary: 5 is best (highest total spend).
   782→    """
   783→    df = rfm_df.copy()
   784→
   785→    df["R_Score"] = pd.qcut(
   786→        df["Recency"].rank(method="first"),
   787→        q=5,
   788→        labels=[5, 4, 3, 2, 1],
   789→    ).astype(int)
   790→
   791→    df["F_Score"] = pd.qcut(
   792→        df["Frequency"].rank(method="first"),
   793→        q=5,
   794→        labels=[1, 2, 3, 4, 5],
   795→    ).astype(int)
   796→
   797→    df["M_Score"] = pd.qcut(
   798→        df["Monetary"].rank(method="first"),
   799→        q=5,
   800→        labels=[1, 2, 3, 4, 5],
   801→    ).astype(int)
   802→
   803→    df["RFM_Score_Code"] = (
   804→        df["R_Score"].astype(str) + df["F_Score"].astype(str) + df["M_Score"].astype(str)
   805→    )
   806→    df["RFM_Score_Total"] = df["R_Score"] + df["F_Score"] + df["M_Score"]
   807→
   808→    return df
   809→
   810→
   811→def assign_rfm_segments(df: pd.DataFrame) -> pd.DataFrame:
   812→    """Maps customer R, F, and M scores to actionable business segment labels."""
   813→    df = df.copy()
   814→
   815→    def get_segment(row) -> str:
   816→        r = row["R_Score"]
   817→        f = row["F_Score"]
   818→        m = row["M_Score"]
   819→
   820→        if r >= 4 and f >= 4 and m >= 4:
   821→            return "Champions"
   822→        if r <= 2 and f >= 4 and m >= 4:
   823→            return "Can't Lose Them"
   824→        if r >= 3 and f >= 3 and m >= 3:
   825→            return "Loyal Customers"
   826→        if r <= 2 and (f >= 3 or m >= 3):
   827→            return "At Risk"
   828→        if r >= 4 and f in [2, 3] and m >= 2:
   829→            return "Potential Loyalists"
   830→        if r >= 4 and f == 1:
   831→            return "Recent Customers"
   832→        if r == 3 and f in [1, 2] and m in [2, 3]:
   833→            return "Promising"
   834→        if r in [2, 3] and f in [2, 3] and m in [2, 3]:
   835→            return "Customers Needing Attention"
   836→        if r in [2, 3] and f <= 2 and m <= 2:
   837→            return "About to Sleep"
   838→        if r == 1 and f == 1 and m == 1:
   839→            return "Lost"
   840→        if r <= 2:
   841→            return "Hibernating"
   842→        return "Promising"
   843→
   844→    df["RFM_Segment"] = df.apply(get_segment, axis=1)
   845→    return df
   846→
   847→
   848→def run_rfm_pipeline(
   849→    cleaned_df: pd.DataFrame,
   850→    output_path: Optional[Path] = RFM_DATA_FILE,
   851→) -> Tuple[pd.DataFrame, pd.DataFrame]:
   852→    """Executes end-to-end RFM metric generation, scoring, and segmentation."""
   853→    rfm = calculate_rfm_metrics(cleaned_df)
   854→    rfm_scored = score_rfm_quantiles(rfm)
   855→    rfm_final = assign_rfm_segments(rfm_scored)
   856→
   857→    total_rev = rfm_final["Monetary"].sum()
   858→    total_cust = len(rfm_final)
   859→
   860→    summary = (
   861→        rfm_final.groupby("RFM_Segment")
   862→        .agg(
   863→            CustomerCount=("CustomerID", "count"),
   864→            TotalRevenue=("Monetary", "sum"),
   865→            AvgRecency=("Recency", "mean"),
   866→            AvgFrequency=("Frequency", "mean"),
   867→            AvgMonetary=("Monetary", "mean"),
   868→            AvgAOV=("AOV", "mean"),
   869→        )
   870→        .reset_index()
   871→    )
   872→
   873→    summary["CustomerSharePct"] = (summary["CustomerCount"] / total_cust * 100).round(2)
   874→    summary["RevenueSharePct"] = (summary["TotalRevenue"] / total_rev * 100).round(2)
   875→    summary["AvgRecency"] = summary["AvgRecency"].round(1)
   876→    summary["AvgFrequency"] = summary["AvgFrequency"].round(1)
   877→    summary["AvgMonetary"] = summary["AvgMonetary"].round(2)
   878→    summary["AvgAOV"] = summary["AvgAOV"].round(2)
   879→    summary["TotalRevenue"] = summary["TotalRevenue"].round(2)
   880→
   881→    summary = summary.sort_values("TotalRevenue", ascending=False).reset_index(drop=True)
   882→
   883→    if output_path:
   884→        output_path = Path(output_path)
   885→        output_path.parent.mkdir(parents=True, exist_ok=True)
   886→        rfm_final.to_csv(output_path, index=False)
   887→        print(f"RFM analysis completed: {len(rfm_final):,} customers segmented and saved to {output_path}")
   888→
   889→    return rfm_final, summary
   890→
   891→
   892→if __name__ == "__main__":
   893→    from src.data_processing import load_raw_data, clean_transactions
   894→    raw_df = load_raw_data()
   895→    cleaned_df, _ = clean_transactions(raw_df)
   896→    rfm_df, segment_summary = run_rfm_pipeline(cleaned_df)
   897→    print("\n--- RFM Segment Summary ---")
   898→    print(segment_summary.to_string(index=False))
   899→```

   903→### 4. [`src/clustering.py`](file:///c:/Customer%20Segmentation%20&%20RFM%20Analysis/src/clustering.py)
   904→*Log1p skewness removal, StandardScaler, multi-metric cluster evaluation ($k \in [2, 9]$), KMeans model training, PCA decomposition, and persona profiling.*
   905→
   906→```python
   907→"""
   908→Machine Learning Customer Segmentation Module.
   909→Performs skewness transformation, feature scaling, optimal k evaluation (Elbow & Silhouette),
   910→K-Means clustering, PCA dimensionality reduction, and customer persona profiling.
   911→"""
   912→
   913→import sys
   914→from pathlib import Path
   915→from typing import Dict, List, Optional, Tuple
   916→
   917→PROJECT_ROOT = Path(__file__).resolve().parent.parent
   918→if str(PROJECT_ROOT) not in sys.path:
   919→    sys.path.insert(0, str(PROJECT_ROOT))
   920→
   921→import numpy as np
   922→import pandas as pd
   923→from scipy import stats
   924→from sklearn.cluster import KMeans
   925→from sklearn.decomposition import PCA
   926→from sklearn.metrics import calinski_harabasz_score, davies_bouldin_score, silhouette_score
   927→from sklearn.preprocessing import StandardScaler
   928→
   929→from src.config import (
   930→    CLUSTER_METRICS_FILE,
   931→    DEFAULT_OPTIMAL_K,
   932→    K_MAX,
   933→    K_MIN,
   934→    RANDOM_SEED,
   935→    REPORTS_DIR,
   936→    RFM_DATA_FILE,
   937→)
   938→
   939→
   940→def preprocess_rfm_for_clustering(
   941→    rfm_df: pd.DataFrame,
   942→    features: Optional[List[str]] = None,
   943→) -> Tuple[np.ndarray, pd.DataFrame, StandardScaler, Dict]:
   944→    """Applies log transformation (np.log1p) and StandardScaler to RFM features."""
   945→    if features is None:
   946→        features = ["Recency", "Frequency", "Monetary"]
   947→
   948→    raw_features = rfm_df[features].copy()
   949→    initial_skew = {col: round(float(stats.skew(raw_features[col])), 3) for col in features}
   950→
   951→    log_transformed = np.log1p(raw_features)
   952→    post_log_skew = {col: round(float(stats.skew(log_transformed[col])), 3) for col in features}
   953→
   954→    scaler = StandardScaler()
   955→    scaled_array = scaler.fit_transform(log_transformed)
   956→    scaled_df = pd.DataFrame(scaled_array, columns=[f"{col}_scaled" for col in features], index=rfm_df.index)
   957→
   958→    skew_info = {
   959→        "features": features,
   960→        "initial_skewness": initial_skew,
   961→        "post_log_skewness": post_log_skew,
   962→    }
   963→
   964→    return scaled_array, scaled_df, scaler, skew_info
   965→
   966→
   967→def evaluate_cluster_range(
   968→    scaled_features: np.ndarray,
   969→    k_min: int = K_MIN,
   970→    k_max: int = K_MAX,
   971→    random_state: int = RANDOM_SEED,
   972→) -> pd.DataFrame:
   973→    """Evaluates KMeans clustering performance across a range of k values."""
   974→    results = []
   975→
   976→    for k in range(k_min, k_max + 1):
   977→        kmeans = KMeans(
   978→            n_clusters=k,
   979→            init="k-means++",
   980→            n_init=10,
   981→            max_iter=300,
   982→            random_state=random_state,
   983→        )
   984→        labels = kmeans.fit_predict(scaled_features)
   985→
   986→        inertia = kmeans.inertia_
   987→        sil = silhouette_score(
   988→            scaled_features,
   989→            labels,
   990→            sample_size=min(1500, len(scaled_features)),
   991→            random_state=random_state,
   992→        )
   993→        db_index = davies_bouldin_score(scaled_features, labels)

  1244→def simulate_campaign_ab_test(
  1245→    rfm_df: pd.DataFrame,
  1246→    baseline_response_rate: float = 0.032,  # 3.20% standard blanket blast
  1247→    target_lift_pct: float = 15.0,           # Target: +15.0% relative lift
  1248→    sample_split: float = 0.5,               # 50/50 A/B split

  1253→    n_total = len(rfm_df)
  1254→    n_control = int(n_total * sample_split)
  1255→    n_treatment = n_total - n_control
  1256→
  1257→    control_success = np.random.binomial(n=n_control, p=baseline_response_rate)
  1258→    control_rate = control_success / n_control
  1259→
  1260→    targeted_rate_calibrated = baseline_response_rate * (1.0 + (target_lift_pct / 100.0))
  1261→    treatment_success = np.random.binomial(n=n_treatment, p=targeted_rate_calibrated)
  1262→    treatment_rate = treatment_success / n_treatment
  1263→
  1264→    abs_lift = treatment_rate - control_rate
  1265→    rel_lift = (abs_lift / control_rate) * 100.0
  1266→

  1268→    pooled_p = (control_success + treatment_success) / (n_control + n_treatment)
  1269→    se = np.sqrt(pooled_p * (1 - pooled_p) * (1 / n_control + 1 / n_treatment))
  1270→    z_stat = (treatment_rate - control_rate) / se
  1271→    p_val = 2 * (1 - stats.norm.cdf(abs(z_stat)))

  1275→    aov_control = 65.00
  1276→    aov_targeted = 82.50

  1280→    control_revenue = control_success * aov_control
  1281→    control_profit = (control_revenue * margin) - control_cost
  1282→    control_roi = ((control_profit) / control_cost) * 100.0 if control_cost > 0 else 0
  1283→
  1284→    treatment_cost = n_treatment * cpc_email
  1285→    treatment_revenue = treatment_success * aov_targeted
  1286→    treatment_profit = (treatment_revenue * margin) - treatment_cost
  1287→    treatment_roi = ((treatment_profit) / treatment_cost) * 100.0 if treatment_cost > 0 else 0
  1288→

  1556→    for cid in customer_ids:

  1558→        country = customers_dict[cid]["Country"]

  1562→
  1563→        last_order_days_ago = np.random.randint(min_rec, max(min_rec + 1, min(max_rec, total_days)))

  1580→

  1582→            invoice_seq += 1
  1583→            is_cancellation = (np.random.random() < 0.025)

  1594→                else:

  1596→
  1597→                unit_price = round(prod["base_price"] * np.random.uniform(0.95, 1.05), 2)
  1598→
  1599→                records.append({
  1600→                    "InvoiceNo": invoice_no,
  1601→                    "StockCode": prod["code"],
  1602→                    "Description": prod["desc"],
  1603→                    "Quantity": qty,
  1604→                    "InvoiceDate": invoice_dt.strftime("%Y-%m-%d %H:%M:%S"),
  1605→                    "UnitPrice": unit_price,
  1606→                    "CustomerID": cid,
  1607→                    "Country": country,
  1608→                    "Category": prod["cat"],
  1609→                })
  1610→
  1611→            if is_cancellation:
  1612→                cancel_invoice_no = f"C{invoice_no}"

  1615→                    records.append({
  1616→                        "InvoiceNo": cancel_invoice_no,
  1617→                        "StockCode": prod["code"],
  1618→                        "Description": prod["desc"],
  1619→                        "Quantity": -int(np.random.choice([1, 2, 4])),
  1620→                        "InvoiceDate": cancel_dt.strftime("%Y-%m-%d %H:%M:%S"),
  1621→                        "UnitPrice": prod["base_price"],
  1622→                        "CustomerID": cid,
  1623→                        "Country": country,
  1624→                        "Category": prod["cat"],
  1625→                    })
  1626→
  1627→    df = pd.DataFrame(records)
  1628→
  1629→    # Introduce guest checkouts (~8% missing CustomerIDs)
  1630→    missing_mask = np.random.random(len(df)) < 0.08
  1631→    df.loc[missing_mask, "CustomerID"] = np.nan
  1632→
  1633→    promo_indices = np.random.choice(df.index, size=int(len(df) * 0.002), replace=False)
  1634→    df.loc[promo_indices, "UnitPrice"] = 0.0
  1635→
  1636→    if len(df) < n_transactions:
  1637→        extra_needed = n_transactions - len(df)
  1638→        sampled_rows = df.sample(extra_needed, replace=True).copy()
  1639→        for i in range(len(sampled_rows)):
  1640→            sampled_rows.iloc[i, sampled_rows.columns.get_loc("InvoiceNo")] = str(invoice_seq + i)
  1641→        df = pd.concat([df, sampled_rows], ignore_index=True)
  1642→
  1643→    df["InvoiceDate"] = pd.to_datetime(df["InvoiceDate"])
  1644→    df = df.sort_values("InvoiceDate").reset_index(drop=True)
  1645→
  1646→    if output_path:
  1647→        output_path = Path(output_path)
  1648→        output_path.parent.mkdir(parents=True, exist_ok=True)
  1649→        df.to_csv(output_path, index=False)
  1650→        print(f"Generated {len(df):,} transactions across {df['CustomerID'].nunique():,} unique customers.")
  1651→        print(f"Saved raw transaction dataset to: {output_path}")
  1652→
  1653→    return df
  1654→

  1657→    generate_synthetic_transactions()
