   169→# 2. SELF-HEALING DATA LOADER
   170→# ============================================================================
   171→
   172→@st.cache_data(show_spinner=False)
   173→def initialize_and_load_data(_force_refresh: bool = False):
   174→    """If RFM/cluster CSVs are missing, generate data and run the full pipeline.
   175→    On legacy CSVs lacking CLV columns, lazily backfill them."""
   176→    if not RFM_DATA_FILE.exists() or not CLUSTER_METRICS_FILE.exists():
   177→        with st.spinner("🚀 Initializing pipeline: Generating sample transactions, training ML clusters, fitting BG/NBD..."):
   178→            raw_df = generate_synthetic_transactions(n_transactions=50000, n_customers=5000)
   179→            cleaned_df, _ = clean_transactions(raw_df, output_path=CLEANED_DATA_FILE)
   180→            rfm_df, _ = run_rfm_pipeline(cleaned_df, output_path=RFM_DATA_FILE)
   181→            rfm_clustered, cluster_profiles, eval_df, _, _ = run_clustering_pipeline(
   182→                rfm_df, n_clusters=4, output_path=RFM_DATA_FILE
   183→            )
   184→            rfm_clustered, _ = compute_predictive_clv(cleaned_df, rfm_clustered, output_path=RFM_DATA_FILE)
   185→    else:
   186→        rfm_df = pd.read_csv(RFM_DATA_FILE)
   187→        cluster_profiles = pd.read_csv(CLUSTER_METRICS_FILE)
   188→        eval_file = REPORTS_DIR / "clustering_evaluation_metrics.csv"
   189→        eval_df = pd.read_csv(eval_file) if eval_file.exists() else pd.DataFrame()
   190→
   191→        # Self-heal CLV columns if needed
   192→        if "Predicted_CLV_365D" not in rfm_df.columns and CLEANED_DATA_FILE.exists():
   193→            with st.spinner("🔮 Backfilling CLV columns on legacy CSV..."):
   194→                cleaned_df = pd.read_csv(CLEANED_DATA_FILE)
   195→                rfm_df, _ = compute_predictive_clv(cleaned_df, rfm_df, output_path=RFM_DATA_FILE)
   196→
   197→    return rfm_df, cluster_profiles, eval_df
   198→
   199→
   200→# Sidebar: Clear-cache button so users can re-run the pipeline and see fresh data
   201→if st.sidebar.button("🔄 Clear cache & reload from disk"):
   202→    st.cache_data.clear()
   203→    st.rerun()

   206→    rfm_df, cluster_profiles, eval_df = initialize_and_load_data()
